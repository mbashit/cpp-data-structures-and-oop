cmake_minimum_required(VERSION 3.16)
project(cpp_data_structures_and_oop CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

file(GLOB DIRS LIST_DIRECTORIES true "${CMAKE_SOURCE_DIR}/*")
foreach(dir ${DIRS})
  if(IS_DIRECTORY ${dir} AND NOT dir MATCHES "build$|\\.git$")
    file(GLOB srcs "${dir}/*.cpp")
    if(srcs)
      get_filename_component(name ${dir} NAME)
      add_executable(${name} ${srcs})
      target_include_directories(${name} PRIVATE ${dir})
    endif()
  endif()
endforeach()
